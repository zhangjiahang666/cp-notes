专题 · 图论

图的相关概念 · 图的存储（邻接表 / 链式前向星）· 图的遍历（DFS / BFS / 0-1 BFS）

一、图的基本概念

作用

图是二元组 G=(V,E)：V 为点集（顶点/节点），E 为边集；点数 |V| 称为图的阶。用来描述「对象之间的关系」这种结构。

什么时候用：需要描述网络、关系、路径、连接问题的时候。

常用概念与术语

分类说明
无向图边是无序对 (u,v)
有向图边是有序对 (u,v)，写作 u→v
混合图 / 赋权图同时含两种边 / 边带权值
简单图 / 多重图无自环且无重边 / 允许自环、重边（竞赛默认允许，需注意题意）
稀疏图 / 稠密图边数远小于 |V|² / 接近 |V|²（决定算法选择）

度数（Degree）

无向图中 d(v) 为与 v 相连的边数；有向图中分出度 d+(v) 与入度 d−(v)。d(v)=0 孤立点，d(v)=1 叶节点。

握手定理：无向图 ∑d(v) = 2|E|（每条边贡献 2 度）⟹ 奇点数量一定为偶数，常用作判错。

路径与连通性

术语含义
路径 Path / 回路 Circuit不重复点的迹 / 起点等于终点的迹
环 Cycle起点=终点且中间不重复点的回路（简单回路）
连通分量（无向）极大连通子图；连通图=任意两点连通
强连通分量（有向）任意两点互相可达的极大子图

补图：相同点集、补上所有缺失边的图；反图：有向图所有边方向反转。

二、图的存储

方式实现复杂度适用
直接存边结构体数组存 (u,v,w)遍历 O(m)Kruskal 等边操作
邻接矩阵g[u][v] 存权（0/1/∞）查询 O(1)、遍历 O(n²)稠密图、n≤2000
邻接表vector<pair> adj[u]O(n+m)最常用，简单好写
链式前向星head/to/w/nxt 数组O(n+m)性能极致、竞赛常用

邻接表（推荐）

#include <bits/stdc++.h>
using namespace std;
const int N = 100005;
vector<pair<int,int>> adj[N];   // adj[u]: (终点 v, 边权 w)

int main() {
  int n, m; cin >> n >> m;
  for (int i = 1; i <= m; i++) {
    int u, v, w; cin >> u >> v >> w;
    adj[u].push_back({v, w});
    adj[v].push_back({u, w});   // 无向图加双向
  }
  for (int u = 1; u <= n; u++)
    for (auto [v, w] : adj[u])
      cout << u << " -> " << v << " (" << w << ")\n";
  return 0;
}

链式前向星

#include <bits/stdc++.h>
using namespace std;
const int N = 100005, M = 200005;
struct Edge { int to, w, nxt; } e[M];
int head[N], tot;

void add(int u, int v, int w) {   // 头插法加边
  e[++tot] = {v, w, head[u]};
  head[u] = tot;
}

int main() {
  int n, m; cin >> n >> m;
  for (int i = 1; i <= m; i++) {
    int u, v, w; cin >> u >> v >> w;
    add(u, v, w);
  }
  for (int u = 1; u <= n; u++)
    for (int i = head[u]; i; i = e[i].nxt)
      cout << u << " -> " << e[i].to << " (" << e[i].w << ")\n";
  return 0;
}
邻接矩阵遍历是 O(n²)，大图一定用邻接表或前向星，才能做到线性 O(n+m)。

三、图的遍历

DFS 深度优先搜索

递归访问每个未访问的邻接点，vis[] 防重复。适用于连通块遍历、图染色、拓扑排序、强连通分量、回溯搜索。

bool vis[N];
void dfs(int u) {
  vis[u] = true;
  for (int i = head[u]; i; i = e[i].nxt) {
    int v = e[i].to;
    if (!vis[v]) dfs(v);
  }
}

BFS 广度优先搜索

按层扩展（队列），由于层次性天然求无权图最短路径：dis[v]=dis[u]+1，pre[] 记录前驱用于路径还原。

int dis[N], pre[N];

void bfs(int s) {
  queue<int> q;
  q.push(s); vis[s] = true;
  dis[s] = 0; pre[s] = -1;
  while (!q.empty()) {
    int u = q.front(); q.pop();
    for (int i = head[u]; i; i = e[i].nxt) {
      int v = e[i].to;
      if (!vis[v]) {
        vis[v] = true;
        dis[v] = dis[u] + 1;
        pre[v] = u;
        q.push(v);
      }
    }
  }
}

void restore(int x) {            // 还原 s -> x 的路径
  vector<int> res;
  for (int v = x; v != -1; v = pre[v]) res.push_back(v);
  reverse(res.begin(), res.end());
  for (int v : res) cout << v << ' ';
}

0-1 BFS 与优先队列 BFS

变体适用核心复杂度
0-1 BFS边权只有 0/10 权入队首、1 权入队尾，队内代价单调O(n+m)
优先队列 BFS状态图最短路最小堆取总代价最小节点，首次出队才处理O((n+m)log n)
 
// 0-1 BFS 模板（边权 0 或 1）
deque<int> dq;
dq.push_front(s); dis[s] = 0;
while (!dq.empty()) {
  int u = dq.front(); dq.pop_front();
  for (auto [v, w] : adj[u]) {
    if (dis[v] > dis[u] + w) {
      dis[v] = dis[u] + w;
      if (w == 0) dq.push_front(v);   // 0 权放队首
      else        dq.push_back(v);    // 1 权放队尾
    }
  }
}
速记：DFS 走到底再回头（回溯、方案数）；BFS 一层一层推（最短路径、最少步数）；0-1 BFS 是带权 BFS 的轻量版；边权任意正数则上 Dijkstra。

四、Dijkstra 最短路

求单源最短路：从起点 s 出发，到图上所有点的最短距离。边权必须非负。思想：每次从未确定的点中挑距离最小的点加入集合，用它松弛邻边。

朴素 Dijkstra（稠密图，O(n²)）

#include <bits/stdc++.h>
using namespace std;
const int N = 1005, INF = 1e9;
int g[N][N], dis[N];
bool vis[N];

void dijkstra(int s, int n) {
    for (int i = 1; i <= n; i++) dis[i] = INF;
    dis[s] = 0;
    for (int i = 1; i <= n; i++) {
        int u = -1;
        for (int j = 1; j <= n; j++)
            if (!vis[j] && (u == -1 || dis[j] < dis[u])) u = j;
        vis[u] = true;
        for (int v = 1; v <= n; v++)
            if (dis[v] > dis[u] + g[u][v])
                dis[v] = dis[u] + g[u][v];
    }
}

堆优化 Dijkstra（稀疏图，O((n+m) log n)）

#include <bits/stdc++.h>
using namespace std;
typedef pair<int,int> pii;
const int N = 100005, INF = 1e9;
vector<pii> adj[N];
int dis[N];
bool vis[N];

void dijkstra(int s) {
    fill(dis, dis + N, INF);
    dis[s] = 0;
    priority_queue<pii, vector<pii>, greater<pii>> pq;  // 小根堆 (距离, 点)
    pq.push({0, s});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        for (auto [v, w] : adj[u])
            if (dis[v] > dis[u] + w) {
                dis[v] = dis[u] + w;
                pq.push({dis[v], v});
            }
    }
}
复杂度选型：n≤1000 用朴素版 O(n²)；n≥10⁵ 用堆优化 O((n+m)log n)。边权有负数不能用 Dijkstra，改 SPFA。

五、并查集

维护「不相交集合」：快速判断两个元素是否在同一集合、合并两个集合。核心操作 find（找祖先）+ union（合并），加路径压缩后接近 O(1)。

核心思想

每个集合用一棵树表示，根节点是「集合代表」。find(x) 沿着父指针一直走到根，返回根编号；union(a,b) 就是把 a 的根挂到 b 的根下面。

路径压缩

find 时把一路上经过的节点直接指向根，下次再查就一步到位。这是并查集快的关键。

#include <bits/stdc++.h>
using namespace std;
const int N = 100005;
int fa[N];

int find(int x) {
    if (fa[x] != x) fa[x] = find(fa[x]);  // 路径压缩
    return fa[x];
}

void unite(int a, int b) {
    a = find(a); b = find(b);
    if (a != b) fa[a] = b;
}

int main() {
    int n, m; cin >> n >> m;
    for (int i = 1; i <= n; i++) fa[i] = i;   // 初始化：自己是自己的爹
    while (m--) {
        int op, a, b; cin >> op >> a >> b;
        if (op == 1) unite(a, b);              // 合并
        else cout << (find(a) == find(b) ? "Yes" : "No") << "\n";
    }
    return 0;
}
